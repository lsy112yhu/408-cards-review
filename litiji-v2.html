1经典处理：用辅助数组统计元素出现次数（3 号讲义 p1）讲义原码

方法页（非真题）：当数组元素取值范围已知（如 0 ≤ a[i] ≤ 8）时，用辅助数组 b 以元素值作下标直接计数。示例：int a[6] = {2, 8, 2, 1, 3, 8}。若只需判断是否出现过，把计数改为置 1（b[a[i]] = 1）。

for (int i = 0; i < n; i++) {
    b[a[i]]++;
}

这是后面主元素（上篇 2）、未现最小正整数（上篇 3）、删绝对值重复结点（下篇 3）几题的共同套路：值域不大时，辅助数组直数就是最稳的通用解。

使用前提：辅助数组 b 必须先清零、长度须覆盖元素值域（如值域 0~8 需至少 9 个位置），否则统计会读到未初始化的值。复杂度：时间 O(n)，辅助空间 O(U)（U 为值域大小）。

22013 统考真题：找主元素（3 号讲义 p5–p11）讲义原码

已知整数序列 A = (a0, …, an-1)，其中 0 ≤ ai < n。若存在 x 出现次数 m > n/2，则称 x 为主元素（如 (0,5,5,3,5,7,5,5) 的主元素是 5）。设计算法找出主元素，存在则输出该元素，否则输出 -1。（三段式要求，下同略）

讲义给了两种统计策略，都是辅助数组直数。解法一：边统计边判断，一旦某元素计数超过 n/2 立即输出。

void func(int A[], int n) {
    int *temp = (int*)malloc(sizeof(int) * n);
    for (int i = 0; i < n; i++) temp[i] = 0;
    for (int i = 0; i < n; i++) {
        temp[A[i]]++;
        if (temp[A[i]] > n/2) {
            printf("%d\n", A[i]);
            return;
        }
    }
    printf("%d\n", -1);
}

解法二：先全部统计完，再遍历辅助数组找计数过半者（此时下标 i 即元素值）。

void func(int A[], int n) {
    int *temp = (int*)malloc(sizeof(int) * n);
    for (int i = 0; i < n; i++) temp[i] = 0;
    for (int i = 0; i < n; i++) temp[A[i]]++;
    for (int i = 0; i < n; i++) {
        if (temp[i] > n/2) {
            printf("%d\n", i);
            return;
        }
    }
    printf("%d\n", -1);
}

复杂度：时间 O(n)，空间 O(n)。讲义另提到摩尔投票法（O(1) 空间）为更优解，截图未给代码，按口径不收。

原码疑似（审核记录，不改原码）：两种解法的 temp 在返回前均未释放，存在内存泄漏。

32018 统考真题：未出现的最小正整数（3 号讲义 p12–p23）讲义原码

给定含 n（n ≥ 1）个整数的数组，找出数组中未出现的最小正整数。例如 {-5, 3, 2, 3} → 1；{1, 2, 3} → 4。

讲义给了辅助数组 n+2、n+1、n 三个大小版本，思路相同，此处收最干净的 n+1 版：标记 1..n 中出现过的数，第一个没标记的就是答案；全出现则答案是 n+1。

int func(int a[], int n) {
    int *temp = (int*)malloc(sizeof(int) * (n + 1));
    for (int i = 0; i < n + 1; i++) temp[i] = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] > 0 && a[i] < n + 1) temp[a[i]] = 1;
    }
    for (int i = 1; i <= n; i++) {
        if (temp[i] == 0) return i;
    }
    return n + 1;
}

复杂度：时间 O(n)，空间 O(n)。讲义要点另提到本题的纯暴力写法（从 1 到 n+1 逐个回数组里找，两层循环 O(n²)），纯暴力版附在本题之后。原码疑似（审核记录，不改原码）：原码的 temp 在返回路径上未释放。

附：本题的纯暴力写法【此段代码为整理者补写，非讲义原码】（讲义要点提及、未给代码）——从 1 到 n+1 逐个候选值、每个都回数组里线性查找，两层循环 O(n²)：

int func(int a[], int n) { // A3 的纯暴力写法（讲义要点提及、未给代码，整理者补写）
    for (int x = 1; x <= n + 1; x++) {
        bool found = false;
        for (int i = 0; i < n; i++)
            if (a[i] == x) { found = true; break; }
        if (!found) return x;
    }
    return -1;
}

42011 统考真题：两个等长升序序列的中位数（4 号讲义 p6–p13；题面见 3 号 p4）讲义原码

长度为 L 的升序序列 S，处在第 ⌈L/2⌉ 个位置的数称为 S 的中位数。两个序列的中位数是含它们所有元素的升序序列的中位数（如 S1=(11,13,15,17,19)、S2=(2,4,6,8,20)，中位数是 11）。现有两个等长升序序列 A 和 B，求它们的中位数。原题要求在时间和空间两方面都尽可能高效；下面的解法一是保分通用解，达不到原题最佳效率（最优为二分法，按口径未收）。

解法一：把两个序列整个归并到 temp 里，直接取下标 L-1 处（即合并后第 L 个）。讲义明确定位这是通用直接解，并明确说二分最优解不推荐。

int func(int S1[], int S2[], int L) {
    int *temp = (int*)malloc(sizeof(int) * 2 * L);
    int i = 0, j = 0, k = 0;
    while (i < L && j < L) {
        if (S1[i] < S2[j]) temp[k++] = S1[i++];
        else temp[k++] = S2[j++];
    }
    while (i < L) temp[k++] = S1[i++];
    while (j < L) temp[k++] = S2[j++];
    return temp[L - 1];
}

解法二：同一思路的省空间版——不存数组，归并时计数，数到第 L 个立即返回（时间 O(L)，空间 O(1)）。

int func(int S1[], int S2[], int L) {
    int i = 0, j = 0, count = 0, ans = 0;
    while (i < L && j < L) {
        if (S1[i] < S2[j]) ans = S1[i++];
        else ans = S2[j++];
        count++;
        if (count == L) return ans;
    }
    return -1;
}

注：解法二讲义定稿代码在一个序列先走完时以 return -1 收尾，未写处理剩余元素的循环——正常输入下数满 L 个必然在循环内返回，-1 不会触达，照讲义原样收录。

口径说明（审核意见）：解法二带省空间优化色彩，按最严口径应移出正文；考虑到它与解法一同为归并顺序取数的通用思路、并非二分最优解，暂保留并在此标注。原码疑似（审核记录）：解法一的 temp 未释放。

52020 统考真题：三元组最小距离（3 号讲义 p12、p25 仅题面）暴力解：整理者补写

定义三元组 (a, b, c) 的距离 D = |a-b| + |b-c| + |c-a|。给定 3 个非空整数集合 S1、S2、S3，按升序分别存储在 3 个数组中。计算并输出所有可能三元组中的最小距离。如 S1={-1,0,9}、S2={-25,-10,10,11}、S3={2,9,17,30,41}，最小距离为 2（三元组 (9,10,9)）。

暴力解：三重循环枚举全部三元组，逐个算距离取最小。不用管三个数组是否有序。

int func(int S1[], int S2[], int S3[], int n1, int n2, int n3) {
    int ans = INT_MAX;
    for (int i = 0; i < n1; i++)
        for (int j = 0; j < n2; j++)
            for (int k = 0; k < n3; k++) {
                int d = abs(S1[i] - S2[j]) + abs(S2[j] - S3[k]) + abs(S3[k] - S1[i]);
                if (d < ans) ans = d;
            }
    return ans;
}

复杂度：时间 O(n1·n2·n3)，空间 O(1)。已用题面示例验算：输出 2 ✓。注：按统考惯例用 int；若数据范围极大、距离之和可能溢出 int，可整体换成 long long（审核建议，枚举思路不变）。

6综合应用题：把所有奇数移动到所有偶数前边（7 号讲义 p1 仅题面）暴力解：整理者补写

已知线性表按顺序存储，且每个元素都是不相同的整数型元素，设计把所有奇数移动到所有偶数前边的算法。（原题要求时间最短、辅助空间最小，指向双指针划分；暴力解不满足该要求，定位保分打法。）

void func(int A[], int n) {
    int *temp = (int*)malloc(sizeof(int) * n);
    int k = 0;
    for (int i = 0; i < n; i++)
        if (A[i] % 2 != 0) temp[k++] = A[i]; //第一遍：先放奇数
    for (int i = 0; i < n; i++)
        if (A[i] % 2 == 0) temp[k++] = A[i]; //第二遍：再放偶数
    for (int i = 0; i < n; i++) A[i] = temp[i];
    free(temp);
}

复杂度：时间 O(n)，空间 O(n)。

7综合应用题：找出数组中第 k 小的元素（7 号讲义 p1 题面、p6–p10 优化解）暴力解：整理者补写

试编写一个算法，使之能够在数组 L[1...n] 中找出第 k 小的元素（从小到大排序后处于第 k 个位置的元素）。

讲义给的是借快排划分只递归一侧的快速选择（平均 O(n)），已剔除。暴力解：先全排序，再直接取第 k 个——排序用最简单的简单选择排序即可。

int func(int L[], int n, int k) { //数组下标 0..n-1，k 从 1 开始数
    for (int i = 0; i < n - 1; i++) { //简单选择排序，全排升序
        int min = i;
        for (int j = i + 1; j < n; j++)
            if (L[j] < L[min]) min = j;
        if (min != i) { int t = L[i]; L[i] = L[min]; L[min] = t; }
    }
    return L[k - 1];
}

复杂度：时间 O(n²)，空间 O(1)。前置条件：1 ≤ k ≤ n（题面默认成立，k 越界会访问越界）。已验算 ✓。

82016 统考真题：集合划分（7 号讲义 p1 题面、p14–p17 优化解；8 号 p1 残段同题）暴力解：整理者补写

已知由 n（n ≥ 2）个正整数构成的集合 A = {ak | 0 ≤ k < n}，将其划分为两个不相交的子集 A1、A2，元素个数分别是 n1、n2，元素之和分别是 S1、S2。设计划分算法，满足 |n1 − n2| 最小且 |S1 − S2| 最大，并返回 S2 − S1。

讲义要点明说「全排序是本题的暴力解」：全排升序后，前 ⌊n/2⌋ 个最小的归 A1、其余归 A2，个数差最小、和差自然最大。讲义的快速选择解法已剔除。

int func(int A[], int n) {
    for (int i = 0; i < n - 1; i++) { //简单选择排序，全排升序
        int min = i;
        for (int j = i + 1; j < n; j++)
            if (A[j] < A[min]) min = j;
        if (min != i) { int t = A[i]; A[i] = A[min]; A[min] = t; }
    }
    int s1 = 0, s2 = 0;
    for (int i = 0; i < n / 2; i++) s1 += A[i];
    for (int i = n / 2; i < n; i++) s2 += A[i];
    return s2 - s1;
}

复杂度：时间 O(n²)（瓶颈在全排序），空间 O(1)。已用讲义手绘示例 {10,20,…,70}（n=7）验算：返回 160 ✓。

注：返回 S2 − S1 是沿用讲义同题原码的接口约定；元素之和在统考规模下 int 足够，范围极大时可换 long long 防溢出。

9综合应用题：荷兰国旗问题（7 号讲义 p1 仅框架）暴力解：整理者补写

设有一个仅由红、白、蓝三种颜色条块组成的序列，存储在一个顺序表中，编写算法使条块按红、白、蓝顺序排好（原题要求时间复杂度 O(n)，指向三指针一趟划分；暴力解为计数重填）。题目给出的类型与函数框架：typedef enum {RED, WHITE, BLUE} color; void Flag_Arrange(color a[], int n)。

void Flag_Arrange(color a[], int n) {
    int cntR = 0, cntW = 0;
    for (int i = 0; i < n; i++) { //第一遍：只数个数
        if (a[i] == RED) cntR++;
        else if (a[i] == WHITE) cntW++;
    }
    for (int i = 0; i < n; i++) { //第二遍：按个数整段重填
        if (i < cntR) a[i] = RED;
        else if (i < cntR + cntW) a[i] = WHITE;
        else a[i] = BLUE;
    }
}

复杂度：时间 O(n)，空间 O(1)。思路就是上篇 1 的计数套路用在三色上。

10顺序表习题组（删除 / 互换 / 查找类）（4 号讲义 p1 仅题面）暴力解：整理者补写

03：对长度为 n 的顺序表 L，删除其中所有值为 x 的数据元素。
04：从顺序表中删除值在 s 与 t 之间（包含 s、t，要求 s < t）的所有元素。
05：从有序顺序表中删除所有值重复的元素，使表中元素值各不相同。
07：一位数组 A[m+n] 中依次存放两个线性表 (a1,…,am) 和 (b1,…,bn)，将两个顺序表的位置互换，即把 (b1,…,bn) 放到 (a1,…,am) 前面。
08：线性表元素递增有序且顺序存储，用最少时间查找值为 x 的元素：找到则与后继元素交换位置，找不到则插入并保持递增有序。
09：给定三个长度均为 n、无重复元素的递增序列 A、B、C，逐行输出同时存在于三个序列中的所有元素（如 A={1,2,3}、B={2,3,4}、C={-1,0,2}，输出 2）。

03、04、05 用同一个最笨的套路：逐个扫描，命中就整体左移删掉，不用紧凑计数法。以下设顺序表为王道 SqList（data 数组＋length）。

// 03：删除所有值为 x 的元素
void func(SqList &L, ElemType x) {
    int i = 0;
    while (i < L.length) {
        if (L.data[i] == x) {
            for (int j = i; j < L.length - 1; j++) L.data[j] = L.data[j + 1];
            L.length--;
        } else i++;
    }
}
// 04：删除值在 [s, t] 内的元素（与 03 只差判断条件）
void func(SqList &L, ElemType s, ElemType t) {
    int i = 0;
    while (i < L.length) {
        if (L.data[i] >= s && L.data[i] <= t) {
            for (int j = i; j < L.length - 1; j++) L.data[j] = L.data[j + 1];
            L.length--;
        } else i++;
    }
}
// 05：有序表删除重复元素（有序则重复必相邻）
void func(SqList &L) {
    int i = 0;
    while (i < L.length - 1) {
        if (L.data[i] == L.data[i + 1]) {
            for (int j = i + 1; j < L.length - 1; j++) L.data[j] = L.data[j + 1];
            L.length--;
        } else i++;
    }
}

07 暴力解：另开一个数组，先抄第二个表、再抄第一个表，最后抄回。

void func(int A[], int m, int n) {
    int *temp = (int*)malloc(sizeof(int) * (m + n));
    for (int i = 0; i < n; i++) temp[i] = A[m + i];
    for (int i = 0; i < m; i++) temp[n + i] = A[i];
    for (int i = 0; i < m + n; i++) A[i] = temp[i];
    free(temp);
}

08 暴力解：线性查找（不用二分），找到就与后继交换；没找到就找插入位置、整体右移腾位插入。

void func(SqList &L, ElemType x) {
    int i = 0;
    while (i < L.length && L.data[i] < x) i++; //从头逐个找
    if (i < L.length && L.data[i] == x) {
        if (i + 1 < L.length) { //与后继交换（已是最后元素则无后继，不动）
            ElemType t = L.data[i]; L.data[i] = L.data[i + 1]; L.data[i + 1] = t;
        }
    } else { //没找到：i 处即插入位置，右移腾位
        if (L.length >= MaxSize) return; //表已满，插入失败
        for (int j = L.length; j > i; j--) L.data[j] = L.data[j - 1];
        L.data[i] = x;
        L.length++;
    }
}

09 暴力解：逐个枚举 A 的元素，分别去 B、C 里线性查找（两层嵌套），都在才输出（不用三指针同步推进的优化写法）。

void func(int A[], int B[], int C[], int n) {
    for (int i = 0; i < n; i++) {
        bool inB = false, inC = false;
        for (int j = 0; j < n; j++) if (B[j] == A[i]) { inB = true; break; }
        for (int j = 0; j < n; j++) if (C[j] == A[i]) { inC = true; break; }
        if (inB && inC) printf("%d\n", A[i]);
    }
}

复杂度：03/04/05 时间最坏 O(n²)（瓶颈在逐个左移，无删除时仅 O(n)）；07 时间 O(m+n)、空间 O(m+n)；08 时间 O(n)；09 时间 O(n²)。

11数组循环左移与逆置（2 号讲义 p1 仅要点提及）暴力解：整理者补写

要点提及的两道常见题：将数组元素循环左移 p 位；逆置数组（顺序表）元素。讲义未给题面与代码，下按要点所述方法补写。

// 循环左移 p 位：辅助数组存前 p 个，其余整体前移，再把暂存的放到尾部（要点原述方法）
void func(int A[], int n, int p) {
    int *temp = (int*)malloc(sizeof(int) * p);
    for (int i = 0; i < p; i++) temp[i] = A[i];
    for (int i = 0; i < n - p; i++) A[i] = A[i + p];
    for (int i = 0; i < p; i++) A[n - p + i] = temp[i];
    free(temp);
}
// 逆置：下标两端逐对交换——数组逆置最朴素的写法就是它本身
void func(int A[], int n) {
    for (int i = 0; i < n / 2; i++) {
        int t = A[i]; A[i] = A[n - 1 - i]; A[n - 1 - i] = t;
    }
}

左移的前置条件：0 ≤ p ≤ n（题面默认）。

12快速排序：Partition ＋ QuickSort（6 号讲义 p1、p4 王道版）讲义原码

实现快速排序。板书要点：左右双指针、以第一个元素作枢轴、初始方向从右往左。演示数组 6, 1, 5, 9, 8, 4, 7。（讲师另有 int 数组自写版，同题从略。）

本题为算法实现题，不是解题用的暴力解。复杂度：时间 O(n log n)，辅助空间 O(n)（另有递归栈开销）。原码疑似（审核记录，不改原码）：全局辅助数组 B 的声明依赖未在代码中展示的 n 定义与初始化，使用时须先按实际规模分配好 B。

本题为算法实现题（题目本身即实现快速排序），不是解题用的暴力解，凭此注与标签区分。复杂度：平均时间 O(n log n)、最坏 O(n²)；递归栈平均 O(log n)、最坏 O(n)。

int Partition(ElemType A[], int low, int high) { //一趟划分
    ElemType pivot = A[low]; //将当前表中第一个元素设为枢轴
    while (low < high) {
        while (low < high && A[high] >= pivot) --high;
        A[low] = A[high]; //将比枢轴小的元素移动到左端
        while (low < high && A[low] <= pivot) ++low;
        A[high] = A[low]; //将比枢轴大的元素移动到右端
    }
    A[low] = pivot; //枢轴元素存放到最终位置
    return low;
}

void QuickSort(ElemType A[], int low, int high) {
    if (low < high) {
        int pivotpos = Partition(A, low, high); //划分
        QuickSort(A, low, pivotpos - 1); //依次对两个子表进行递归排序
        QuickSort(A, pivotpos + 1, high);
    }
}

13归并排序：Merge ＋ MergeSort（5 号讲义 p1、p7–p9 王道版）讲义原码

实现递归二路归并排序。示例：初始关键字 [49] [38] [65] [97] [76] [13] [27]，归并后为 13 27 38 49 65 76 97。（讲师另有 int 数组自写版，同题从略。）

ElemType *B = (ElemType*)malloc((n + 1) * sizeof(ElemType)); //辅助数组 B
void Merge(ElemType A[], int low, int mid, int high) {
    //A 的两段 A[low...mid] 和 A[mid+1...high] 各自有序，合并成一个有序表
    int i, j, k;
    for (k = low; k <= high; k++) B[k] = A[k]; //将 A 中所有元素复制到 B 中
    for (i = low, j = mid + 1, k = i; i <= mid && j <= high; k++) {
        if (B[i] <= B[j]) A[k] = B[i++]; //将较小值复制到 A 中
        else A[k] = B[j++];
    }
    while (i <= mid) A[k++] = B[i++]; //若第一个表未检测完，复制
    while (j <= high) A[k++] = B[j++]; //若第二个表未检测完，复制
}

void MergeSort(ElemType A[], int low, int high) {
    if (low < high) {
        int mid = (low + high) / 2; //从中间划分两个子序列
        MergeSort(A, low, mid); //对左侧子序列进行递归排序
        MergeSort(A, mid + 1, high); //对右侧子序列进行递归排序
        Merge(A, low, mid, high); //归并
    }
}

14合并两个有序顺序表 / 有序数组（4 号讲义 p1–p5 王道版＋讲师数组版）讲义原码

将两个有序顺序表合并为一个新的有序顺序表，并由函数返回结果顺序表（王道习题 06）。

bool Merge(SeqList A, SeqList B, SeqList &C) {
    //将有序顺序表 A 与 B 合并为一个新的有序顺序表 C
    if (A.length + B.length > C.maxSize) return false; //大于顺序表的最大长度
    int i = 0, j = 0, k = 0;
    while (i < A.length && j < B.length) { //循环，两两比较，小者存入结果表
        if (A.data[i] <= B.data[j]) C.data[k++] = A.data[i++];
        else C.data[k++] = B.data[j++];
    }
    while (i < A.length) C.data[k++] = A.data[i++]; //还剩一个没有比较完的顺序表
    while (j < B.length) C.data[k++] = B.data[j++];
    C.length = k;
    return true;
}

    while (i < m && j < n) {
        if (a[i] < b[j]) temp[k++] = a[i++];
        else temp[k++] = b[j++];
    }
    while (i < m) temp[k++] = a[i++];
    while (j < n) temp[k++] = b[j++];
}

152021 统考真题：读代码题 cmpCountSort（比较计数排序）（8 号讲义 p1–p2）讲义原码＋整理者试算

已知排序算法如下（代码见下），回答：1）若 int a[] = {25, -10, 25, 10, 11, 19}，调用 cmpCountSort(a, b, 6) 后数组 b 的内容是什么？2）a 中有 n 个元素时，元素之间的比较次数是多少？3）该算法稳定吗？若否，修改为稳定排序算法。

void cmpCountSort(int a[], int b[], int n) {
    int i, j, *count;
    count = (int*)malloc(sizeof(int) * n);
    for (i = 0; i < n; i++) count[i] = 0;
    for (i = 0; i < n - 1; i++)
        for (j = i + 1; j < n; j++)
            if (a[i] < a[j]) count[j]++;
            else count[i]++;
    for (i = 0; i < n; i++) b[count[i]] = a[i];
    free(count);
}

这道题的给定代码本身就是最朴素的两两枚举计数（每个元素数一遍「有几个元素比我小」，count 即其排序名次），与暴力解同源，故收录于此。整理者按代码逐步试算：

1）各元素 count 依次为 {5, 0, 4, 1, 2, 3}，故 b = {-10, 10, 11, 19, 25, 25}。

2）比较次数 = 双重循环的执行次数 = n(n-1)/2。

复杂度：时间 O(n²)，计数辅助空间 O(n)。

3）不稳定：两个 25 中，前者（a[0]）count=5、后者（a[2]）count=4，相对次序被颠倒。改法：把判断改为 if (a[i] <= a[j]) count[j]++; else count[i]++;——相等时给后者计数，使前者名次靠前，次序得以保持。

12009 统考真题：查找单链表倒数第 k 个结点（10 号讲义 p32–p43）讲义原码

带头结点的单链表只给出头指针 list。在不改变链表的前提下，查找倒数第 k 个位置上的结点（k 为正整数）：成功则输出该结点 data 并返回 1，否则返回 0。

讲义共给了 4 种解法，其中两种是暴力解（快慢指针两种优化写法已剔除）。暴力解一：数组法——先求长，再把链表整个搬进辅助数组，按下标 n-k 直取。

int Length(LinkList list) {
    int len = 0;
    LNode *p = list;
    while (p->next != NULL) {
        p = p->next;
        len++;
    }
    return len;
}

void ListArray(LinkList list, int temp[], int n) {
    LNode *p = list->next;
    for (int i = 0; i < n; i++) {
        temp[i] = p->data;
        p = p->next;
    }
}

int func(LinkList list, int k) {
    int n = Length(list);
    if (k > n) return 0;
    int *temp = (int*)malloc(sizeof(int) * n);
    ListArray(list, temp, n);
    printf("%d", temp[n - k]);
    return 1;
}

暴力解二：先求长再正数定位——倒数第 k 个就是正数第 n-k+1 个，从头逐个往后数（讲义代码注释里原标注「//暴力解」；求长函数在本页写作 LenList）。

int func(LinkList list, int k) { //暴力解
    int n = LenList(list);

    int count = 0;
    while (p->next != NULL) {
        p = p->next;
        count++;
        if (count == n - k + 1) break;
    }
    printf("%d", p->data);
    return 1;
}

void func(LinkList head, int n) {

        }
    }
}

综合 01：在带头结点的单链表 L 中，删除所有值为 x 的结点并释放空间（值为 x 的结点不唯一）。
综合 02：在带头结点的单链表 L 中删除一个最小值结点（假设最小值结点唯一）。

// 综合 01：删所有值为 x 的结点——盯着前驱看后继，命中就摘
void func(LinkList L, ElemType x) {
    LNode *p = L;
    while (p->next != NULL) {
        if (p->next->data == x) {
            LNode *q = p->next;
            p->next = q->next;
            free(q);
        } else p = p->next;
    }
}
// 综合 02：删最小值结点——先完整遍历找到最小值结点的前驱，再删
void func(LinkList L) {
    if (L->next == NULL) return; //空表无可删
    LNode *pre = L, *minpre = L, *p = L->next;
    ElemType min = p->data; //假设链表非空
    while (p != NULL) {
        if (p->data < min) { min = p->data; minpre = pre; }
        pre = p;
        p = p->next;
    }
    LNode *q = minpre->next;
    minpre->next = q->next;
    free(q);
}

7快慢指针四类应用的暴力对应版（9 号讲义仅板书思路；2 号要点）暴力解：整理者补写

9 号讲义以「快慢指针」总览了四类链表应用：倒数第 k 个结点、链表中点、两链表交点、判环。讲义只给技巧思路、无代码。其中倒数第 k 个与交点的暴力解已分别见下篇 1、下篇 4；这里补中点与判环两题的暴力版。

找中点——暴力解：先求表长，再从头走 (n-1)/2 步：奇数取正中、偶数取中间两结点中靠前的一个，与讲义板书口径一致（空表返回 NULL）。

LNode* func(LinkList L) {
    int n = 0;
    for (LNode *p = L->next; p != NULL; p = p->next) n++;
    if (n == 0) return NULL;
    LNode *p = L->next;
    for (int i = 0; i < (n - 1) / 2; i++) p = p->next;
    return p;
}

判环——暴力解：把走过的每个结点地址存进数组（满了就翻倍扩容），每到一个新结点就回头逐个比对地址，重复出现即有环（不用快慢指针追及）。

bool func(LinkList L) {
    int cap = 16, n = 0;
    LNode **seen = (LNode**)malloc(sizeof(LNode*) * cap);
    for (LNode *p = L->next; p != NULL; p = p->next) {
        for (int i = 0; i < n; i++)
            if (seen[i] == p) { free(seen); return true; }
        if (n == cap) {
            cap *= 2;
            seen = (LNode**)realloc(seen, sizeof(LNode*) * cap);
        }
        seen[n++] = p;
    }
    free(seen);
    return false;
}

复杂度：中点时间 O(n)（两遍）；判环时间 O(n²)、空间 O(n)。两段均已编译验算 ✓。

9尾插法 / 头插法建立单链表（12 号讲义 p2–p3 教材代码）讲义原码

依次读入整数、以 9999 为结束标志、带头结点，分别用尾插法（正向）与头插法（逆向）建立单链表。头插法读入顺序与表中顺序相反，可用来实现链表逆置。

LinkList List_TailInsert(LinkList &L) { //正向建立单链表
    int x; //设元素类型为整型
    L = (LNode*)malloc(sizeof(LNode)); //创建头结点
    LNode *s, *r = L; //r 为表尾指针
    scanf("%d", &x);
    while (x != 9999) { //输入 9999 表示结束
        s = (LNode*)malloc(sizeof(LNode));
        s->data = x;
        r->next = s;
        r = s; //r 指向新的表尾结点
        scanf("%d", &x);
    }
    r->next = NULL; //尾结点指针置空
    return L;
}

LinkList List_HeadInsert(LinkList &L) { //逆向建立单链表
    LNode *s; int x;
    L = (LNode*)malloc(sizeof(LNode)); //创建头结点
    L->next = NULL; //初始为空链表
    scanf("%d", &x);
    while (x != 9999) {
        s = (LNode*)malloc(sizeof(LNode)); //创建新结点
        s->data = x;
        s->next = L->next;
        L->next = s; //将新结点插入表中
        scanf("%d", &x);
    }
    return L;
}

10单链表就地逆置（12 号讲义 p4–p5，王道综合题 03）讲义原码

试编写算法将带头结点的单链表就地逆置（辅助空间复杂度 O(1)）。方法即把元素结点逐个摘下、用头插法重新挂回去。

LinkList Reverse_1(LinkList L) {
    LNode *p, *r; //p 为工作指针，r 为 p 的后继，以防断链
    p = L->next; //从第一个元素结点开始
    L->next = NULL; //先将头结点 next 域置为 NULL
    while (p != NULL) { //依次将元素结点摘下
        r = p->next; //暂存 p 的后继
        p->next = L->next; //将 p 结点插入到头结点之后
        L->next = p;
        p = r;
    }
    return L;
}

11单链表基本操作四件套（10 号讲义）讲义原码

结点类型定义（讲义各页通用）：typedef struct LNode { ElemType data; struct LNode *next; } LNode, *LinkList;

来源细分：Length、ListInsert、ListDelete 为讲义原码；GetElem 的函数体为整理者补写（讲义截图残缺），已在代码内注释标明。

// 求表长（p17 写法一；p19 另有从头结点后继起、判 p 本身为空的写法二，同题从略）
int Length(LinkList L) {
    int len = 0; //计数变量
    LNode *p = L;
    while (p->next != NULL) {
        p = p->next;
        len++; //每访问一个结点，计数加 1
    }
    return len;
}

// 按序号查找 GetElem（截图只拍到函数头，函数体被页面截断；以下按王道通行写法补全，非讲义原码）
LNode *GetElem(LinkList L, int i) {
    if (i < 0) return NULL; //位序非法（i=0 时返回头结点）
    LNode *p = L; //指针 p 指向当前扫描到的结点
    int j = 0; //头结点是第 0 个结点
    while (p != NULL && j < i) {
        p = p->next;
        j++;
    }
    return p; //i 大于表长时返回 NULL
}

// 按位序插入 ListInsert（p27）
bool ListInsert(LinkList &L, int i, ElemType e) {
    LNode *p = L;
    int j = 0;
    while (p != NULL && j < i - 1) { //循环找到第 i-1 个结点
        p = p->next;
        j++;
    }
    if (p == NULL) return false; //i 值不合法
    LNode *s = (LNode*)malloc(sizeof(LNode));
    s->data = e;
    s->next = p->next; //①
    p->next = s;       //②（①②顺序不能颠倒）
    return true;
}

// 按位序删除 ListDelete（p31）
bool ListDelete(LinkList &L, int i, ElemType &e) {
    LNode *p = L;
    int j = 0;
    while (p->next != NULL && j < i - 1) { //循环找到第 i-1 个结点
        p = p->next;
        j++;
    }
    if (p->next == NULL || j > i - 1) return false; //i 值不合法
    LNode *q = p->next; //令 q 指向被删除结点
    e = q->data; //用 e 返回元素的值
    p->next = q->next; //将 *q 结点从链中"断开"
    free(q); //释放结点的存储空间
    return true;
}

双链表在 p 之后插入 s 的代码片段（p32；命题追踪：2023）：① s->next = p->next；② p->next->prior = s；③ s->prior = p；④ p->next = s。语句顺序不唯一，但 ① 必须在 ④ 之前。

原码疑似（审核记录，不改原码）：① ListInsert 在 i ≤ 0 时循环不执行、会直接在表头后插入，未拦截非法位序；② 双链表插入片段中 p->next->prior = s 要求 p 有后继结点，p 为尾结点时不适用（命题语境默认 p 非尾结点）。

1递归入门：求阶乘（5 号讲义 p2）讲义原码

用函数求 n 的阶乘，讲义给了递归与循环两种写法（同为按定义直算，无技巧）。本题与顺序表、链表均无关，是讲义的递归入门例题，单列于附篇。

// 递归版
int f(int n) {
    if (n == 1) return 1; //递归出口
    return n * f(n - 1);
}

// 循环版
int f(int n) {
    int a = 1;
    for (int i = 1; i <= n; i++) {
        a = a * i;
    }
    return a;
}

原码疑似（审核记录，不改原码）：递归版终止条件只有 n==1，n=0 或负数时无法正常结束；循环版 n=0 返回 1，两版适用范围不一致；int 在 13! 即溢出。复杂度：递归版时间 O(n)、栈空间 O(n)；循环版时间 O(n)、空间 O(1)。
